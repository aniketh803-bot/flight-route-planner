const $ = id => document.getElementById(id);
    const graph = Object.fromEntries(Object.keys(airports).map(code => [code, []]));

for (const [a, b] of edges) {

let maxStops = 1;

const number = value => Math.round(value).toLocaleString();
    const radians = degrees => degrees * Math.PI / 180;

function distanceBetween(a, b) {
      const p = airports[a];
      const q = airports[b];
      const dLat = radians(q.lat - p.lat);
      const dLon = radians(q.lon - p.lon);
      const h = Math.sin(dLat / 2) ** 2 +
        Math.cos(radians(p.lat)) * Math.cos(radians(q.lat)) *
        Math.sin(dLon / 2) ** 2;

return 6371 * 2 * Math.atan2(
        Math.sqrt(h),
        Math.sqrt(Math.max(0, 1 - h))
      );
    }

function getRoute(path) {
      let distance = 0;
      for (let i = 1; i < path.length; i++) {
        distance += distanceBetween(path[i - 1], path[i]);
      }
      const legs = path.length - 1;
      const stops = legs - 1;
      const minutes = Math.round(distance / 820 * 60 + legs * 30 + stops * 90);
      return { path, distance, stops, minutes };
    }

function formatTime(minutes) {
      return `${Math.floor(minutes / 60)}h ${minutes % 60}m`;
    }

function findRoutes(origin, destination, limit) {
      const found = [];

function visit(path) {
        const last = path[path.length - 1];
        if (last === destination) {
          found.push(getRoute(path));
          return;
        }
        if (path.length - 1 >= limit + 1) return;

for (const next of graph[last]) {
          if (!path.includes(next)) visit([...path, next]);
        }
      }

visit([origin]);
      return found;
    }

function populateAirports() {
      const sorted = Object.entries(airports)
        .sort((a, b) => a[1].city.localeCompare(b[1].city));

for (const [code, airport] of sorted) {
        for (const id of ["origin", "destination"]) {
          const option = document.createElement("option");
          option.value = code;
          option.textContent = `${airport.city} (${code})`;
          $(id).appendChild(option);
        }
      }

$("origin").value = "LHR";
      $("destination").value = "SIN";
    }

function project(code) {
      const { lon, lat } = airports[code];
      return {
        x: 35 + (lon + 180) / 360 * 870,
        y: 30 + (80 - lat) / 140 * 360
      };
    }

function svgElement(tag, attributes, text) {
      const element = document.createElementNS("http://www.w3.org/2000/svg", tag);
      for (const [key, value] of Object.entries(attributes)) {
        element.setAttribute(key, value);
      }
      if (text !== undefined) element.textContent = text;
      return element;
    }

function arc(a, b) {
      const p = project(a);
      const q = project(b);
      const lift = Math.min(85, Math.hypot(q.x - p.x, q.y - p.y) * .23);
      return `M ${p.x} ${p.y} Q ${(p.x + q.x) / 2} ${(p.y + q.y) / 2 - lift} ${q.x} ${q.y}`;
    }

function updateZoom() {

      );
      $("zoom-out").disabled = zoom <= 1;
      $("zoom-in").disabled = zoom >= 2.5;
    }

function drawMap(route) {
      const layer = $("map-layer");
      layer.replaceChildren();

for (let lon = -150; lon <= 150; lon += 30) {
        const x = 35 + (lon + 180) / 360 * 870;
        layer.appendChild(svgElement("line", {
          x1: x, y1: 25, x2: x, y2: 397,
          stroke: "#304158", "stroke-width": .6, opacity: .45
        }));
      }

for (let lat = -30; lat <= 60; lat += 30) {
        const y = 30 + (80 - lat) / 140 * 360;
        layer.appendChild(svgElement("line", {
          x1: 25, y1: y, x2: 915, y2: y,
          stroke: "#304158", "stroke-width": .6, opacity: .45

function selectRoute(index, announce = true) {
      selectedIndex = index;
      const route = currentRoutes[index];

document.querySelectorAll(".route-card").forEach((card, i) => {
        card.setAttribute("aria-pressed", String(i === index));
      });

$("export").disabled = !route;

        $("distance").textContent = "—";
        $("duration").textContent = "—";
        $("connections").textContent = "—";
        $("map-heading").textContent = "Sample airport network";
        drawMap(null);
        return;
      }

$("distance").innerHTML = `${number(route.distance)} <small>km</small>`;
      $("duration").textContent = formatTime(route.minutes);
      $("connections").textContent = route.stops === 0 ? "Nonstop" : String(route.stops);

      const last = route.path[route.path.length - 1];
      $("map-heading").textContent = `${airports[first].city} → ${airports[last].city}`;
      $("map-svg-desc").textContent =
        `Selected illustrative route: ${route.path.join(" to ")}. ` +
        `Approximate distance ${number(route.distance)} kilometers.`;

drawMap(route);

        $("status").textContent =
          `Selected ${route.path.join(" to ")}. ` +

      }
    }

function renderResults(total) {
      const container = $("routes");
      container.replaceChildren();

        ? `Top 3 of ${total} options`

        empty.className = "panel empty";

        selectRoute(-1, false);

      }

      });

    }

function planRoute(event) {
      if (event) event.preventDefault();

const origin = $("origin").value;
      const destination = $("destination").value;
      $("error").hidden = true;

if (origin === destination) {
        $("error").textContent = "Choose two different airports to plan a route.";
        $("error").hidden = false;
        $("destination").focus();
        return;
      }

const priority = $("priority").value;
      const results = findRoutes(origin, destination, maxStops);

results.sort((a, b) => {
        if (priority === "stops") return a.stops - b.stops || a.distance - b.distance;
        if (priority === "time") return a.minutes - b.minutes || a.distance - b.distance;
        return a.distance - b.distance || a.stops - b.stops;
      });

currentRoutes = results.slice(0, 3);
      zoom = 1;
      renderResults(results.length);

$("status").textContent =
        `${results.length} sample routes found from ${airports[origin].city} ` +
        `to ${airports[destination].city}. ` +
        (results.length ? "The best matching route is selected." : "Try allowing more connections.");
    }
